# ==============================
# 10 Wellness Protocols Scoring
# ==============================

# Mapping symptoms to protocols
symptom_map = {
    # Digestive Health
    "Bloating": ["Parasite", "Leaky Gut", "Gallbladder"],
    "Excess gas": ["Parasite", "Leaky Gut"],
    "Constipation": ["Parasite", "Leaky Gut", "Gallbladder", "Whole Body Detox"],
    "Diarrhea": ["Parasite", "Leaky Gut", "Whole Body Detox"],
    "Stomach cramps": ["Parasite", "Leaky Gut"],
    "Undigested food in stool": ["Parasite", "Leaky Gut"],
    
    # Energy & Cognition
    "Morning fatigue": ["Parasite", "Adrenal", "Anemia"],
    "Afternoon energy crashes": ["Parasite", "Adrenal", "Anemia"],
    "Brain fog": ["Parasite", "Adrenal", "Heavy Metal", "Whole Body Detox"],
    "Frequent headaches": ["Heavy Metal", "Liver", "Kidney"],

    # Skin & Allergies
    "Rashes or hives": ["Parasite", "Leaky Gut", "Whole Body Detox"],
    "Eczema": ["Parasite", "Leaky Gut", "Whole Body Detox"],
    "Itchy skin": ["Parasite", "Leaky Gut"],
    "Seasonal allergies": ["Heavy Metal", "Whole Body Detox"],

    # Sleep & Mood
    "Trouble falling asleep": ["Adrenal", "Menopause"],
    "Waking 1–3 AM": ["Adrenal", "Menopause"],
    "Anxiety or irritability": ["Adrenal", "Menopause"],

    # Appetite & Weight
    "Sugar cravings": ["Parasite", "Adrenal"],
    "Difficulty gaining weight": ["Parasite", "Anemia", "Leaky Gut"],

    # Pain & Physical Function
    "Joint or muscle pain": ["Heavy Metal", "Whole Body Detox", "Kidney"],
    
    # Exposure History
    "Travel to tropical/low sanitation areas": ["Parasite"],
    "Raw or undercooked meat/fish": ["Parasite"],
    "Exposure to heavy metals": ["Heavy Metal"],
    "Medications or alcohol": ["Liver"]
}

# List of all protocols
protocols = ["Parasite", "Leaky Gut", "Adrenal", "Heavy Metal", "Whole Body Detox",
             "Liver", "Kidney", "Gallbladder", "Menopause", "Anemia"]

# Initialize scores
protocol_scores = {p: 0 for p in protocols}

# ==========================
# Example Client Responses
# Replace this list with actual client selections
# ==========================
client_responses = [
    "Bloating",
    "Afternoon energy crashes",
    "Brain fog",
    "Rashes or hives",
    "Trouble falling asleep",
    "Sugar cravings",
    "Travel to tropical/low sanitation areas"
]

# Tally points for each protocol
for symptom in client_responses:
    if symptom in symptom_map:
        for protocol in symptom_map[symptom]:
            protocol_scores[protocol] += 1

# Sort protocols by score (descending)
sorted_protocols = sorted(protocol_scores.items(), key=lambda x: x[1], reverse=True)

# Assign High / Medium / Low
highest_score = sorted_protocols[0][1] if sorted_protocols else 0

def grade(score, highest):
    if score == highest:
        return "High"
    elif score >= highest / 2:
        return "Medium"
    else:
        return "Low"

# Pick top 2 protocols
primary_protocol = sorted_protocols[0]
secondary_protocol = sorted_protocols[1]

# Output
print(f"Primary Protocol: {primary_protocol[0]} - {grade(primary_protocol[1], highest_score)}")
print(f"Secondary Protocol: {secondary_protocol[0]} - {grade(secondary_protocol[1], highest_score)}")

# Optional: Show full scoring (for debugging or reporting)
print("\nAll Protocol Scores:")
for p, score in sorted_protocols:
    print(f"{p}: {score} ({grade(score, highest_score)})")
